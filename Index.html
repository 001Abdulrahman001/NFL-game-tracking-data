import pandas as pd
import numpy as np

def calculate_agility_conversion(combine_tracking, game_tracking):
    """
    Calculates the Agility Conversion Index (ACI) between isolated 
    Combine drill data and NFL regular season tracking data.
    """
    # 1. Extract peak acceleration in Combine Shuttle & 3-Cone
    combine_accel = combine_tracking.groupby('player_id')['a'].max()
    
    # 2. Extract top directional cut velocity from game tracking
    game_cuts = game_tracking[game_tracking['dir_change_deg'] > 45]
    game_burst = game_cuts.groupby('player_id')['s'].quantile(0.90)
    
    # 3. Calculate conversion index ratio
    aci_df = pd.DataFrame({
        'combine_max_accel': combine_accel,
        'game_cut_velocity': game_burst
    })
    
    aci_df['ACI'] = (aci_df['game_cut_velocity'] / aci_df['combine_max_accel']) * 100
    return aci_df.fillna(0)
